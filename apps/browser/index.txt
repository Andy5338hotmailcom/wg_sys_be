    <button onclick="goHome()">🏠</button>
    <input id="addr" placeholder="输入网址" onkeydown="if(event.key==='Enter')navigate()">
    <button onclick="navigate()">前往</button>

<div id="frameWrap">
    <iframe id="webview" src="about:blank"></iframe>
</div>

<script>
const iframe = document.getElementById("webview");
const addrInput = document.getElementById("addr");
let historyStack = [];
let historyIndex = -1;
const homeUrl = "https://www.baidu.com";

// 导航
function navigate(){
    let url = addrInput.value.trim();
    if(!url) return;
    if(!/^https?:\/\//.test(url)){
        url = "https://"+url;
    }
    historyStack.splice(historyIndex+1);
    historyStack.push(url);
    historyIndex = historyStack.length -1;
    iframe.src = url;
}
function goBack(){
    if(historyIndex>0){
        historyIndex--;
        iframe.src = historyStack[historyIndex];
        addrInput.value = historyStack[historyIndex];
    }
}
function goForward(){
    if(historyIndex < historyStack.length-1){
        historyIndex++;
        iframe.src = historyStack[historyIndex];
        addrInput.value = historyStack[historyIndex];
    }
}
function reload(){
    iframe.src = iframe.src;
}
function goHome(){
    addrInput.value = homeUrl;
    navigate();
}
// 初始化主页
addrInput.value = homeUrl;
navigate();
</script>
